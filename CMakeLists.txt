cmake_minimum_required(VERSION 3.21)
project(Omadash VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_AUTOMOC ON)
find_package(Qt6 6.6 REQUIRED COMPONENTS Widgets Test Network WebEngineWidgets PdfWidgets Sql)
add_library(omadash_core src/mailstore.cpp src/google.cpp src/attachments.cpp src/theme.cpp src/localcache.cpp src/keyring.cpp src/mime.cpp src/mailsync.cpp src/gmailactions.cpp)
target_include_directories(omadash_core PUBLIC src)
target_link_libraries(omadash_core PUBLIC Qt6::Widgets Qt6::Network Qt6::Sql)
qt_add_resources(omadash_core fixtures PREFIX "/" FILES src/sample.json)
add_library(omadash_ui src/window.cpp src/window_daily.cpp src/emailview.cpp)
target_link_libraries(omadash_ui PUBLIC omadash_core Qt6::WebEngineWidgets Qt6::PdfWidgets)
add_executable(omadash src/main.cpp)
target_link_libraries(omadash PRIVATE omadash_ui)
enable_testing()
add_executable(omadash_tests tests/test_native.cpp)
target_link_libraries(omadash_tests PRIVATE omadash_ui Qt6::Test)
add_test(NAME native-workflows COMMAND omadash_tests -platform offscreen)
set_tests_properties(native-workflows PROPERTIES ENVIRONMENT "OMADASH_TEST_MODE=1")
install(TARGETS omadash RUNTIME DESTINATION bin)
# Keep the application optimized; avoid expensive optimization of test scaffolding.
target_compile_options(omadash_tests PRIVATE -O0)
add_executable(omadash_daily_tests tests/test_daily.cpp)
target_link_libraries(omadash_daily_tests PRIVATE omadash_ui Qt6::Test)
target_compile_options(omadash_daily_tests PRIVATE -O0)
add_test(NAME daily-regressions COMMAND omadash_daily_tests -platform offscreen)
set_tests_properties(daily-regressions PROPERTIES ENVIRONMENT "OMADASH_TEST_MODE=1")
